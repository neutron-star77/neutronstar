## 10. ⚠️ 避坑清单

| 坑 | 原因 | 解决方案 |
|----|------|----------|
| Tailwind 4 编译报错 | v4 用 `@layer` 不用 `@tailwind` | 确保 `design-tokens.css` 在 `@layer base` 内 |
| CSS 变量在 Tailwind 中不可用 | Tailwind 需显式声明 | 在 `tailwind.config` 中映射 CSS 变量 |
| `astro:transitions` 导致 FOUC | 脚本执行时机问题 | 用 `is:inline` 的内联脚本优先执行 |
| 字体加载闪烁 | Google Fonts 网络延迟 | 用 `font-display: swap` 或自托管 |
| `localStorage` 在 SSR 报错 | 服务端无 window | 用 `is:inline` 包裹，或判 `typeof window` |
| PDF.js worker 路径 404 | Vite 不自动复制静态资源 | 手动放 `public/pdfjs/` 目录 |

<script>
  // 性能检测：仅多核设备启用
  if (navigator.hardwareConcurrency > 4) {
    initStarField();
  }

  function initStarField() {
    const canvas = document.getElementById('hud-starfield');
    if (!canvas) return;

    const ctx = canvas.getContext('2d');
    let width, height, stars, animationId;

    function resize() {
      width = canvas.width = window.innerWidth;
      height = canvas.height = window.innerHeight;
    }

    function createStars(count) {
      return Array.from({ length: count }, () => ({
        x: Math.random() * width,
        y: Math.random() * height,
        radius: Math.random() * 1.5 + 0.3,
        speed: Math.random() * 0.3 + 0.05,
        alpha: Math.random(),
        alphaSpeed: (Math.random() - 0.5) * 0.02,
        color: Math.random() > 0.8 ? '#FF2E97' : '#00E5FF',
      }));
    }

    function draw() {
      ctx.clearRect(0, 0, width, height);

      stars.forEach((star) => {
        // 移动
        star.y += star.speed;
        if (star.y > height) {
          star.y = 0;
          star.x = Math.random() * width;
        }

        // 闪烁
        star.alpha += star.alphaSpeed;
        if (star.alpha > 1 || star.alpha < 0.1) {
          star.alphaSpeed *= -1;
        }

        // 绘制
        ctx.beginPath();
        ctx.arc(star.x, star.y, star.radius, 0, Math.PI * 2);
        ctx.fillStyle = star.color;
        ctx.globalAlpha = Math.max(0.1, Math.min(1, star.alpha));
        ctx.fill();

        // 亮星加光晕
        if (star.radius > 1.2) {
          ctx.beginPath();
          ctx.arc(star.x, star.y, star.radius * 3, 0, Math.PI * 2);
          ctx.fillStyle = star.color;
          ctx.globalAlpha = Math.max(0.02, star.alpha * 0.1);
          ctx.fill();
        }
      });

      ctx.globalAlpha = 1;

      // 偶尔画一条流星
      if (Math.random() < 0.002) {
        drawMeteor();
      }

      animationId = requestAnimationFrame(draw);
    }

    function drawMeteor() {
      const x = Math.random() * width;
      const y = Math.random() * height * 0.3;
      const len = 80 + Math.random() * 120;
      const angle = Math.PI / 4 + (Math.random() - 0.5) * 0.3;

      ctx.save();
      ctx.translate(x, y);
      ctx.rotate(angle);
      const gradient = ctx.createLinearGradient(0, 0, len, 0);
      gradient.addColorStop(0, 'rgba(0, 229, 255, 0.8)');
      gradient.addColorStop(0.5, 'rgba(0, 229, 255, 0.2)');
      gradient.addColorStop(1, 'rgba(0, 229, 255, 0)');
      ctx.strokeStyle = gradient;
      ctx.lineWidth = 1.5;
      ctx.beginPath();
      ctx.moveTo(0, 0);
      ctx.lineTo(len, 0);
      ctx.stroke();
      ctx.restore();
    }

    function start() {
      resize();
      stars = createStars(200);
      draw();
    }

    function stop() {
      if (animationId) cancelAnimationFrame(animationId);
    }

    // 启动
    start();

    // 响应式
    window.addEventListener('resize', () => {
      resize();
      stars = createStars(200);
    });

    // View Transitions 兼容
    document.addEventListener('astro:before-swap', stop);
    document.addEventListener('astro:page-load', start);

    // 页面隐藏时暂停
    document.addEventListener('visibilitychange', () => {
      if (document.hidden) stop();
      else start();
    });
  }
</script>

## 7. ⚠️ 避坑清单

| 坑 | 原因 | 解决方案 |
|----|------|----------|
| 扫描线遮挡点击 | z-index 过高 | 确保 `.hud-shell` 及其子元素全部 `pointer-events: none` |
| 星空 Canvas 内存泄漏 | RAF 未在页面切换时取消 | 监听 `astro:before-swap` 取消 RAF |
| 移动端卡顿 | 粒子数过多 | 移动端检测后减半粒子或禁用 |
| 导航栏 3D 效果导致文字模糊 | transform 亚像素 | 使用整数 translateZ 值 |
| 状态条时钟不更新 | DOM 未找到 | 用 `is:inline` 确保脚本立即执行 |
| 四角括号位置偏移 | inset 计算错误 | 用 `inset: var(--space-4)` 统一 |
| Canvas 在高 DPI 屏模糊 | 未适配 devicePixelRatio | 设置 `canvas.width = innerWidth * dpr` |
| 页面切换后星空消失 | View Transitions 替换 DOM | 在 `astro:page-load` 中重新初始化 |
| 暗角过深影响阅读 | opacity 太高 | 控制在 0.4-0.6 之间 |
| 导航栏 z-index 冲突 | 与内容重叠 | 导航栏 z-index > 内容 z-index |

## 10. ⚠️ 避坑清单

| 坑 | 原因 | 解决方案 |
|----|------|----------|
| TXT 中文乱码 | 文件非 UTF-8 | 保存时明确选 UTF-8 无 BOM |
| TXT 全量加载卡顿 | 大文件一次性读取 | 打字机效果分块渲染 |
| PDF.js worker 404 | 路径未配置 | 设置 `GlobalWorkerOptions.workerSrc` |
| PDF 渲染模糊 | 未适配 DPI | canvas 乘 `devicePixelRatio` |
| PDF 跨域报错 | 文件不在同源 | 放 `public/books/` 同域访问 |
| 打字机效果卡 UI | 同步循环阻塞 | 用 `requestAnimationFrame` 分帧 |
| 书架横向滚动不流畅 | 默认滚轮是纵向 | 监听 wheel 事件转横向 |
| 封面图拉伸变形 | 未设 aspect-ratio | 用 `aspect-ratio: 5/7` + `object-fit: cover` |
| 全屏阅读器 z-index 不够 | 被 HUD 覆盖 | 阅读器 z-index > 所有 HUD 层 |
| 返回按钮失效 | SPA 路由问题 | 用 `history.back()` 或 `<a>` 链接 |

// TXT 大文件分块加载（>1MB 时）
async function loadLargeTxt(url) {
  const response = await fetch(url);
  const reader = response.body.getReader();
  const decoder = new TextDecoder('utf-8');
  let buffer = '';
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    // 每 50KB 渲染一次
    if (buffer.length > 50000) {
      appendToTerminal(buffer);
      buffer = '';
    }
  }
  if (buffer) appendToTerminal(buffer);
}

  // 状态文字轮换
  (function () {
    const el = document.getElementById('ai-status');
    const messages = [
      '[AI_CORE: ONLINE]',
      '[SYNCHRONIZING...]',
      '[DATA_STREAM: ACTIVE]',
      '[UPLINK: STABLE]',
    ];
    let idx = 0;
    setInterval(() => {
      idx = (idx + 1) % messages.length;
      el.textContent = messages[idx];
    }, 3000);

  // Canvas 环绕粒子（仅高性能设备）
  if (navigator.hardwareConcurrency > 4) {
    initAIParticles();
  }

  function initAIParticles() {
    const canvas = document.getElementById('ai-particles');
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    let width, height, particles, animationId;

    function resize() {
      const root = document.getElementById('ai-core-root');
      width = canvas.width = root.offsetWidth + 40;
      height = canvas.height = root.offsetHeight + 40;
    }

    function createParticles(count) {
      return Array.from({ length: count }, () => ({
        angle: Math.random() * Math.PI * 2,
        radius: 50 + Math.random() * 60,
        speed: 0.01 + Math.random() * 0.02,
        size: Math.random() * 2 + 0.5,
        color: Math.random() > 0.5 ? '#00E5FF' : '#FF2E97',
      }));
    }

    function draw() {
      ctx.clearRect(0, 0, width, height);
      const cx = width / 2;
      const cy = height / 2;

      particles.forEach(p => {
        p.angle += p.speed;
        const x = cx + Math.cos(p.angle) * p.radius;
        const y = cy + Math.sin(p.angle) * p.radius;

        ctx.beginPath();
        ctx.arc(x, y, p.size, 0, Math.PI * 2);
        ctx.fillStyle = p.color;
        ctx.globalAlpha = 0.6;
        ctx.fill();

        // 连线
        if (p.radius < 70) {
          ctx.beginPath();
          ctx.moveTo(cx, cy);
          ctx.lineTo(x, y);
          ctx.strokeStyle = p.color;
          ctx.globalAlpha = 0.15;
          ctx.lineWidth = 0.5;
          ctx.stroke();
        }
      });

      ctx.globalAlpha = 1;
      animationId = requestAnimationFrame(draw);
    }

    function start() {
      resize();
      particles = createParticles(30);
      draw();
    }

    function stop() {
      if (animationId) cancelAnimationFrame(animationId);
    }

    start();
    window.addEventListener('resize', resize);

    document.addEventListener('astro:before-swap', stop);
    document.addEventListener('astro:page-load', start);
    document.addEventListener('visibilitychange', () => {
      if (document.hidden) stop();
      else start();
    });
  }
</script>

/**
 * HudMusicPlayer.tsx - HUD 风格全局音乐播放器
 * React 岛屿，client:only 懒加载
 * 集成 APlayer + Meting
 */
import { useEffect, useRef, useState } from 'react';

interface Props {
  server: 'netease' | 'tencent' | 'kugou';
  type: 'playlist' | 'song' | 'album';
  id: string;
  fixed?: boolean;
  autoplay?: boolean;
  volume?: number;
}

declare global {
  interface Window {
    APlayer: any;
    MetingJSEmpty?: any;
  }
}

export default function HudMusicPlayer({
  server,
  type,
  id,
  fixed = true,
  autoplay = false,
  volume = 0.7,
}: Props) {
  const containerRef = useRef<HTMLDivElement>(null);
  const [isLoaded, setIsLoaded] = useState(false);
  const [isPlaying, setIsPlaying] = useState(false);
  const [currentTrack, setCurrentTrack] = useState('');
  const [progress, setProgress] = useState(0);

  useEffect(() => {
    let ap: any = null;

    async function init() {
      // 动态导入 APlayer CSS
      const link = document.createElement('link');
      link.rel = 'stylesheet';
      link.href = 'https://cdn.jsdelivr.net/npm/aplayer@1.10.1/dist/APlayer.min.css';
      document.head.appendChild(link);

      // 动态导入 APlayer
      const APlayerModule = await import('aplayer');
      const APlayer = APlayerModule.default;

      // 动态导入 Meting
      const MetingModule = await import('meting');
      const Meting = MetingModule.default || MetingModule;

      if (!containerRef.current) return;

      const m = new Meting({
        server,
        type,
        id,
        api: 'https://api.i-meto.com/meting/api',
      });

      const tracks = await m.get();

      ap = new APlayer({
        container: containerRef.current,
        fixed,
        autoplay,
        volume,
        listFolded: true,
        listMaxHeight: '200px',
        audio: tracks,
        theme: '#00E5FF',
      });

      ap.on('play', () => setIsPlaying(true));
      ap.on('pause', () => setIsPlaying(false));
      ap.on('canplay', () => {
        const t = ap.list.audios[ap.list.index];
        if (t) setCurrentTrack(t.name + ' - ' + t.artist);
      });
      ap.on('timeupdate', () => {
        if (ap.duration) {
          setProgress((ap.currentTime / ap.duration) * 100);
        }
      });

      setIsLoaded(true);
    }

    init().catch(err => console.warn('[Music] Load error:', err));

    return () => {
      if (ap && ap.destroy) ap.destroy();
    };
  }, [server, type, id, fixed, autoplay, volume]);

  return (
    <div className="hud-music-root" data-playing={isPlaying}>
      <div ref={containerRef} className="hud-music-container"></div>

      {/* HUD 均衡器叠加层 */}
      {isLoaded && (
        <div className="hud-eq-overlay" aria-hidden="true">
          <div className="eq-bars">
            {[...Array(5)].map((_, i) => (
              <span key={i} className={`eq-bar ${isPlaying ? 'active' : ''}`} style={{ animationDelay: `${i * 0.15}s` }}></span>
            ))}
          </div>
          <span className="eq-track mono">{currentTrack || '[NO SIGNAL]'}</span>
          <div className="eq-progress">
            <div className="eq-progress-bar" style={{ width: `${progress}%` }}></div>
          </div>
        </div>
      )}
    </div>
  );
}

<script>
  // 活动项高亮 + 阅读进度
  function initToc() {
    const links = document.querySelectorAll('[data-toc-link]');
    const headings = Array.from(links).map(l => document.getElementById(l.dataset.tocLink));
    const progressBar = document.getElementById('toc-progress-bar');

    if (!headings.length) return;

    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        const id = entry.target.id;
        const link = document.querySelector(`[data-toc-link="${id}"]`);
        if (entry.isIntersecting) {
          links.forEach(l => l.classList.remove('active'));
          link?.classList.add('active');
        }
      });
    }, { rootMargin: '-10% 0px -80% 0px' });

    headings.forEach(h => h && observer.observe(h));

    // 阅读进度
    function updateProgress() {
      const scrollTop = window.scrollY;
      const docHeight = document.body.scrollHeight - window.innerHeight;
      const pct = Math.min(100, Math.max(0, (scrollTop / docHeight) * 100));
      if (progressBar) progressBar.style.width = pct + '%';
    }
    window.addEventListener('scroll', updateProgress, { passive: true });
    updateProgress();
  }

  initToc();
  document.addEventListener('astro:page-load', initToc);
</script>

  (function () {
    const el = document.getElementById('typewriter');
    if (!el) return;
    const messages = [
      '星舰图书馆已上线',

    let isDeleting = false;

    function tick() {
      const msg = messages[msgIdx];
      if (isDeleting) {
        el.textContent = msg.substring(0, charIdx--);
        if (charIdx < 0) {
          isDeleting = false;
          msgIdx = (msgIdx + 1) % messages.length;
          setTimeout(tick, 500);
          return;
        }
      } else {
        el.textContent = msg.substring(0, ++charIdx);
        if (charIdx >= msg.length) {
          isDeleting = true;
          setTimeout(tick, 2000);
          return;
        }
      }
      setTimeout(tick, isDeleting ? 30 : 60);
    }
    tick();
  })();

@layer view-transitions {
  /* ===== 扫描线擦除过渡 ===== */
  @keyframes hud-scan-wipe-in {
    0% {
      clip-path: inset(0 0 100% 0);
      opacity: 0;
    }
    100% {
      clip-path: inset(0 0 0 0);
      opacity: 1;
    }
  }

  @keyframes hud-scan-wipe-out {
    0% {
      clip-path: inset(0 0 0 0);
      opacity: 1;
    }
    100% {
      clip-path: inset(100% 0 0 0);
      opacity: 0;
    }
  }

  /* ===== 数据溶解过渡 ===== */
  @keyframes hud-dissolve-in {
    0% {
      opacity: 0;
      filter: blur(8px) brightness(2);
      transform: scale(0.98);
    }
    100% {
      opacity: 1;
      filter: blur(0) brightness(1);
      transform: scale(1);
    }
  }

  @keyframes hud-dissolve-out {
    0% {
      opacity: 1;
      filter: blur(0) brightness(1);
      transform: scale(1);
    }
    100% {
      opacity: 0;
      filter: blur(8px) brightness(2);
      transform: scale(1.02);
    }
  }

  /* ===== 星舰跃迁过渡 ===== */
  @keyframes hud-warp-in {
    0% {
      opacity: 0;
      transform: perspective(800px) rotateX(12deg) scale(0.9);
      filter: blur(4px) contrast(1.2);
    }
    100% {
      opacity: 1;
      transform: perspective(800px) rotateX(0) scale(1);
      filter: blur(0) contrast(1);
    }
  }

  @keyframes hud-warp-out {
    0% {
      opacity: 1;
      transform: perspective(800px) rotateX(0) scale(1);
    }
    100% {
      opacity: 0;
      transform: perspective(800px) rotateX(-12deg) scale(1.1);
      filter: blur(4px);
    }
  }

  /* ===== 应用过渡到主内容 ===== */
  main#main-content {
    view-transition-name: main-content;
  }

  /* 扫描线擦除（默认） */
  ::view-transition-old(main-content) {
    animation: hud-scan-wipe-out 0.4s ease-in-out both;
  }
  ::view-transition-new(main-content) {
    animation: hud-scan-wipe-in 0.4s ease-in-out both;
  }

  /* 数据溶解（用于文章页） */
  body[data-transition="dissolve"] ::view-transition-old(main-content) {
    animation: hud-dissolve-out 0.5s ease both;
  }
  body[data-transition="dissolve"] ::view-transition-new(main-content) {
    animation: hud-dissolve-in 0.5s ease both;
  }

  /* 星舰跃迁（用于书架/全屏阅读器） */
  body[data-transition="warp"] ::view-transition-old(main-content) {
    animation: hud-warp-out 0.6s ease both;
  }
  body[data-transition="warp"] ::view-transition-new(main-content) {
    animation: hud-warp-in 0.6s ease both;
  }

  /* ===== 减少动效 ===== */
  @media (prefers-reduced-motion: reduce) {
    ::view-transition-old(*),
    ::view-transition-new(*) {
      animation: none !important;
      transition: none !important;
    }
  }
}

        with:

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Type check
        run: pnpm astro check

      - name: Build site
        run: pnpm build

      - name: Generate Pagefind index
        run: pnpm pagefind

          urls: |

    needs: build-and-test
    runs-on: ubuntu-latest
